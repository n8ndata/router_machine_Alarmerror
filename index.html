คู่มือขั้นตอนการติดตั้งใน Google Sheets (Setup Guide)

1. เตรียม Google Sheet

สร้างแท็บชื่อ M/C ไว้สำหรับเก็บรายชื่อเครื่องจักรในคอลัมน์ A (เริ่มแถวที่ 2)

2. เปิด Script Editor

ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script ใน Google Sheets ของคุณ

3. วางโค้ด Apps Script

กดปุ่ม "คัดลอกโค้ด" ด้านล่าง แล้วนำไปวางแทนที่ในไฟล์ Code.gs ทั้งหมด แล้วกด บันทึก

4. Deploy เป็น Web App

กด อนุมัติการใช้งาน (Deploy) > การทำให้ใช้งานได้ใหม่ ตั้งค่า Who has access: Anyone แล้วคัดลอก URL มาวางในช่องด้านบน

โค้ด Google Apps Script (Code.gs) - รูปแบบตรงตามภาพต้นแบบ (คอลัมน์ A ถึง H พร้อมเส้นตาราง) 

// =================================================================
// Google Apps Script for ROUTER Machine Alarm & Downtime Sync Engine
// รองรับโครงสร้างคอลัมน์ตรงตามภาพต้นแบบ (A-H) พร้อมใส่เส้นตาราง
// =================================================================

function doGet(e) {
  var action = e ? e.parameter.action : "";
  var callback = e ? e.parameter.callback : "";

  var ss = SpreadsheetApp.getActiveSpreadsheet();

  if (action === "getMachines") {
    var sheet = ss.getSheetByName("M/C");
    var machines = [];
    if (sheet) {
      var lastRow = sheet.getLastRow();
      if (lastRow >= 1) {
        var values = sheet.getRange(1, 1, lastRow, 1).getValues();
        values.forEach(function(row) {
          var val = row[0] ? row[0].toString().trim() : "";
          if (val && val !== "จำนวนเครื่อง/ชื่อ" && val !== "ชื่อเครื่องจักร" && val !== "Machine Name") {
            machines.push(val);
          }
        });
      }
    }
    var output = JSON.stringify({ result: "success", machines: machines });
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + output + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(output).setMimeType(ContentService.MimeType.JSON);
  }

  if (action === "getAlarms" || action === "getAllData") {
    var sheet = ss.getSheetByName("History Data Alarm");
    var events = [];
    if (sheet) {
      var lastRow = sheet.getLastRow();
      if (lastRow >= 2) {
        var data = sheet.getRange(1, 1, lastRow, 8).getValues();
        data.forEach(function(row) {
          var machine = row[0] ? row[0].toString().trim() : "";
          var timestamp = row[1] ? row[1].toString().trim() : "";
          var code = row[2] ? row[2].toString().trim() : "";
          var groupName = row[3] ? row[3].toString().trim() : "";
          var severity = row[4] ? row[4].toString().toUpperCase().trim() : "";
          var downtimeStr = row[5] ? row[5].toString().trim() : "";
          var message = row[6] ? row[6].toString().trim() : "";
          var countVal = Number(row[7]) || 1;

          if (machine && machine !== "ชื่อเครื่องจักร" && code) {
            var dtMin = 0;
            var hours = downtimeStr.match(/([\d.]+)\s*ชม\./);
            var mins = downtimeStr.match(/([\d.]+)\s*นาที/);
            if (hours || mins) {
              dtMin = (hours ? Number(hours[1]) * 60 : 0) + (mins ? Number(mins[1]) : 0);
            } else {
              dtMin = Number(downtimeStr) || 0;
            }

            for (var c = 0; c < countVal; c++) {
              events.push({
                machine: machine,
                timestamp: timestamp,
                code: code,
                groupName: groupName,
                severity: severity || "WARNING",
                downtimeMinutes: dtMin / countVal,
                message: message
              });
            }
          }
        });
      }
    }
    var outputData = JSON.stringify({ result: "success", events: events });
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + outputData + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(outputData).setMimeType(ContentService.MimeType.JSON);
  }

  return ContentService.createTextOutput("ROUTER Sync Engine Ready");
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var payload = JSON.parse((e && e.postData && e.postData.contents) || "{}");
    
    if (payload.action === "submitKPIReport") {
      var reportSheet = ss.getSheetByName("KPI Reports");
      if (!reportSheet) {
        reportSheet = ss.insertSheet("KPI Reports");
        reportSheet.appendRow([
          "วันที่ส่งรายงาน", "เครื่องจักร", "เวลารวมสูญเสีย", "Critical Count", 
          "D1-D2: Problem Description", "D3: Containment", "D4: Root Cause", "D5-D6: Permanent Action"
        ]);
        reportSheet.getRange(1, 1, 1, 8).setBackground("#b4c6e7").setFontWeight("bold");
      }
      
      reportSheet.appendRow([
        payload.timestamp || new Date().toLocaleString("th-TH"),
        payload.machine || "ALL",
        formatMinutesToHoursOrMinutes(payload.totalDowntimeMinutes || 0),
        payload.criticalCount || 0,
        payload.problemDesc || "",
        payload.containment || "",
        payload.rootCause || "",
        payload.permanentAction || ""
      ]);

      return ContentService.createTextOutput(JSON.stringify({
        result: "success",
        message: "บันทึก KPI Report ลงแท็บ KPI Reports สำเร็จ"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    var rawEvents = payload.events || [];
    var TARGET_SHEET_NAME = "History Data Alarm";
    var targetSheet = ss.getSheetByName(TARGET_SHEET_NAME);
    if (!targetSheet) targetSheet = ss.insertSheet(TARGET_SHEET_NAME);

    // กำหนด Headers คอลัมน์ A ถึง H ตรงตามภาพตัวอย่าง
    var headers = [
      "ชื่อเครื่องจักร", "Timestamp ล่าสุด",
      "# รหัส Code", "กลุ่ม Alarm / Fault", "# Severity",
      "เวลารวมสูญเสีย", "รายละเอียด Alarm", "# (จำนวนครั้งที่เกิด)"
    ];

    function cleanText(value, fallback) {
      var text = value === null || value === undefined ? "" : String(value).trim();
      return text || (fallback !== undefined ? fallback : "-");
    }
    function norm(value) {
      return cleanText(value, "").toLowerCase().replace(/\s+/g, " ");
    }
    function makeKey(machine, code, alarmName) {
      return [norm(machine), norm(code), norm(alarmName)].join("|||");
    }
    function machineSort(a, b) {
      return a.localeCompare(b, "en", { numeric: true, sensitivity: "base" });
    }
    function latestTimestamp(a, b) {
      var da = new Date(a), db = new Date(b);
      if (!isNaN(db.getTime()) && (isNaN(da.getTime()) || db > da)) return b;
      return a || b || "-";
    }
    function severityRank(value) {
      var s = cleanText(value, "").toUpperCase();
      return s === "CRITICAL" ? 2 : (s === "WARNING" ? 1 : 0);
    }

    var machineRows = {};

    var incomingEvents = rawEvents.filter(function(ev) {
      var sev = (ev.severity || "").toString().toUpperCase();
      var code = (ev.code || "").toString().trim();
      var msg = (ev.message || "").toString().toLowerCase();
      
      return sev === "CRITICAL" || sev === "WARNING" ||
             code === "04509" || code === "04512" || code === "04513" || code === "03006" || 
             code === "05206" || code === "03502" || code === "03500" ||
             code === "03901" || code === "03900" || code === "03094" ||
             code === "03016" || code === "03004" ||
             msg.indexOf("alarm") !== -1 || msg.indexOf("error") !== -1 ||
             msg.indexOf("sensor") !== -1 || msg.indexOf("interrupted") !== -1 ||
             msg.indexOf("timeout") !== -1 || msg.indexOf("broken") !== -1 ||
             msg.indexOf("lenk") !== -1 || msg.indexOf("limit switch") !== -1 ||
             msg.indexOf("area sensor") !== -1 || msg.indexOf("machine stop") !== -1 ||
             msg.indexOf("clamped") !== -1 || msg.indexOf("spindle") !== -1 ||
             msg.indexOf("tool not") !== -1 || msg.indexOf("stuck") !== -1;
    });

    incomingEvents.forEach(function(item) {
      var machine = cleanText(item.machine || item.machineName, "Tongtai No.10");
      var alarmName = cleanText(item.groupName || item.message, "Unknown Alarm");
      var code = cleanText(item.code, "-");
      var key = makeKey(machine, code, alarmName);

      if (!machineRows[machine]) machineRows[machine] = {};

      if (!machineRows[machine][key]) {
        machineRows[machine][key] = {
          machine: machine,
          timestamp: cleanText(item.timestamp),
          code: code,
          groupName: alarmName,
          severity: cleanText(item.severity, "WARNING"),
          totalDowntime: 0,
          count: 0,
          message: cleanText(item.message)
        };
      }

      var agg = machineRows[machine][key];
      agg.totalDowntime += Number(item.downtimeMinutes || 0) || 0;
      agg.count += 1;
      if (item.timestamp) agg.timestamp = latestTimestamp(agg.timestamp, item.timestamp);
      if (severityRank(item.severity) > severityRank(agg.severity)) agg.severity = item.severity;
      if (item.message) agg.message = item.message;
    });

    var outputRows = [];
    var headerRows = [];
    var machineNames = Object.keys(machineRows).sort(machineSort);

    machineNames.forEach(function(machine) {
      var keys = Object.keys(machineRows[machine]).sort(function(a, b) {
        var ra = machineRows[machine][a], rb = machineRows[machine][b];
        return String(ra.code).localeCompare(String(rb.code), "en", { numeric: true, sensitivity: "base" }) ||
               String(ra.groupName).localeCompare(String(rb.groupName), "th", { sensitivity: "base" });
      });

      if (!keys.length) return;

      var headerRowNumber = outputRows.length + 1;
      outputRows.push([
        "ชื่อเครื่องจักร", "Timestamp ล่าสุด",
        "# รหัส Code", "กลุ่ม Alarm / Fault", "# Severity",
        "เวลารวมสูญเสีย", "รายละเอียด Alarm", "# (จำนวนครั้งที่เกิด)"
      ]);
      headerRows.push({ type: "header", row: headerRowNumber });

      keys.forEach(function(key) {
        var item = machineRows[machine][key];
        outputRows.push([
          item.machine,
          item.timestamp,
          item.code,
          item.groupName,
          item.severity,
          formatMinutesToHoursOrMinutes(item.totalDowntime),
          item.message !== "-" && item.message !== "" ? item.message : item.groupName,
          item.count
        ]);
      });

      outputRows.push(["", "", "", "", "", "", "", ""]);
    });

    targetSheet.clearContents();
    targetSheet.clearFormats();

    if (outputRows.length > 0) {
      var dataRange = targetSheet.getRange(1, 1, outputRows.length, headers.length);
      dataRange.setValues(outputRows);

      // ใส่เส้นตาราง (Borders) ทุกเซลล์ในตารางอัตโนมัติ
      dataRange.setBorder(true, true, true, true, true, true, "#d9d9d9", SpreadsheetApp.BorderStyle.SOLID);

      headerRows.forEach(function(item) {
        if (item.type === "header") {
          targetSheet.getRange(item.row, 1, 1, headers.length)
            .setBackground("#1e6b3b") // สีเขียวเข้มสไตล์ตาราง Google Sheets
            .setFontColor("#ffffff")
            .setFontWeight("bold")
            .setHorizontalAlignment("center");
        }
      });

      targetSheet.getRange(1, 3, outputRows.length, 1).setHorizontalAlignment("center");
      targetSheet.getRange(1, 5, outputRows.length, 1).setHorizontalAlignment("center");
      targetSheet.getRange(1, 6, outputRows.length, 1).setHorizontalAlignment("right");
      targetSheet.getRange(1, 8, outputRows.length, 1).setHorizontalAlignment("center");
    }

    return ContentService.createTextOutput(JSON.stringify({
      result: "success",
      machines: machineNames.length,
      message: "อัปเดตโครงสร้างตารางตรงตามภาพตัวอย่างเรียบร้อยแล้ว"
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function formatMinutesToHoursOrMinutes(totalMin) {
  if (!totalMin || totalMin <= 0) return "0 นาที";
  if (totalMin >= 60) {
    var hours = Math.floor(totalMin / 60);
    var mins = Math.round(totalMin % 60);
    return mins > 0 ? hours + " ชม. " + mins + " นาที" : hours + " ชม.";
  }
  return Math.round(totalMin * 10) / 10 + " นาที";
}
